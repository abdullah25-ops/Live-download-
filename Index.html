SQL -- users & profiles CREATE TABLE users ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), email TEXT UNIQUE, phone TEXT UNIQUE, password_hash TEXT, -- nullable for oauth users is_active BOOLEAN DEFAULT TRUE, created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), updated_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); CREATE TABLE profiles ( user_id UUID PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE, username TEXT UNIQUE NOT NULL, display_name TEXT, bio TEXT, avatar_url TEXT, followers_count INTEGER DEFAULT 0, following_count INTEGER DEFAULT 0, verified BOOLEAN DEFAULT FALSE, blue_tick_assigned_by TEXT, -- admin email blue_tick_assigned_at TIMESTAMP WITH TIME ZONE, created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), updated_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); -- follows CREATE TABLE follows ( follower_id UUID REFERENCES users(id) ON DELETE CASCADE, followee_id UUID REFERENCES users(id) ON DELETE CASCADE, created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), PRIMARY KEY (follower_id, followee_id) ); -- videos CREATE TABLE videos ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES users(id) ON DELETE CASCADE, title TEXT, description TEXT, status TEXT DEFAULT 'processing', -- processing, published, removed visibility TEXT DEFAULT 'public', -- public/unlisted/private s3_key TEXT, -- original hls_master_url TEXT, -- for streaming thumbnail_url TEXT, duration_ms INTEGER, width INTEGER, height INTEGER, views BIGINT DEFAULT 0, likes_count INTEGER DEFAULT 0, comments_count INTEGER DEFAULT 0, created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), updated_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); CREATE INDEX idx_videos_created_at ON videos(created_at); -- likes, comments, saves CREATE TABLE likes ( user_id UUID REFERENCES users(id), video_id UUID REFERENCES videos(id) ON DELETE CASCADE, created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), PRIMARY KEY (user_id, video_id) ); CREATE TABLE comments ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), video_id UUID REFERENCES videos(id) ON DELETE CASCADE, user_id UUID REFERENCES users(id), parent_comment_id UUID REFERENCES comments(id), text TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); CREATE TABLE saves ( user_id UUID REFERENCES users(id), video_id UUID REFERENCES videos(id), created_at TIMESTAMP WITH TIME ZONE DEFAULT now(), PRIMARY KEY (user_id, video_id) ); -- feeds / events (for ML & analytics) CREATE TABLE events ( id BIGSERIAL PRIMARY KEY, user_id UUID, event_type TEXT, -- view, like, comment, follow, watch_complete, gift_sent, etc. metadata JSONB, created_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); -- live rooms CREATE TABLE live_rooms ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), host_user_id UUID REFERENCES users(id), title TEXT, status TEXT DEFAULT 'live', -- live, ended, scheduled started_at TIMESTAMP WITH TIME ZONE, ended_at TIMESTAMP WITH TIME ZONE, sfu_room_id TEXT, -- id in LiveKit or SFU viewer_count INTEGER DEFAULT 0, created_at TIMESTAMP WITH TIME ZONE DEFAULT now() ); Video pipeline & streaming Upload flow: Client requests signed upload URL from Video Service -> backend returns pre-signed S3 URL. Client uploads file (multipart) directly to S3. Backend enqueues video job to Kafka -> transcoder workers pick up job. FFmpeg worker transcodes into adaptive bitrate HLS (e.g., 360p/540p/720p), generate thumbnails, and store assets on S3. Save hls_master_url in DB. Moderation: run AI detection on video (frames + audio) and produce a risk score. If high risk, mark as pending review. Streaming: serve HLS via CDN (CloudFront). Use signed URLs for private/unlisted content. Compression: FFmpeg settings template (CRF & x264/x265) and input-specific scaling. Workers auto-scale via Kubernetes HPA. Live: Use LiveKit (recommended) or mediasoup + custom SFU. LiveKit provides multi-host, recording, guest request flow, and SDKs for mobile/web. Enforce max 8 participants (1 host + 7 guests). Use Redis / presence service for viewer count and chat pub/sub. Record live session to HLS for VOD. Live streaming workflows & guest requests Room created by host (POST /live/rooms) -> SFU room created. Guest request: POST /live/rooms/:id/requests -> host receives notification (via socket). Host accepts -> backend issues a join token for the guest; SFU handles media join. Moderation: moderators can remove participants and mute audio/video; maintain moderator event logs. Moderation & safety AI moderation: On upload, run automated checks (nudity, sexual content, violence, hate speech, illegal content, OCR/text in frames). Analyze audio for abusive language using speech-to-text -> NLP classifier. Score content; automated actions: score < threshold: publish score in warning band: publish with label or limit visibility + notify human moderator score above threshold: auto-hide & push to human review queue Violation policy: First violation: automated warning + restriction (temporary mute from posting or limit reach). Repeated violations: temporally suspend (7/30 days). Severe: permanent ban. Admins can override automated actions after human review. Human admin panel: queue with prioritized items (severity + ML confidence), full playback and context, action buttons, and audit log. Coins, gifts, revenue share & payments Pricing: 40 coins = 20 PKR => 1 coin = 0.5 PKR. Buy coins: User initiates purchase in-app -> backend creates order -> redirect to Easypaisa/JazzCash / PSP -> callback/webhook from PSP -> credits coins to wallet. Persist transactions in coin_transactions. Gift sending: Sender wallet debited, receiver credited with coins_amount * (1 - platform_commission). Commission = 40% -> receiver gets 60% coins value (or coins credited and later cashout redeemable). Record gift_event. Animated gift overlay via client-side Lottie/Canvas animations. Assets served from CDN. Withdrawals: Creator requests withdrawal -> create withdrawals row with status pending. Admin approves via Admin Panel -> backend triggers payment to Easypaisa/JazzCash via provider API. Platform takes 40% commission at gift redemption time or during withdrawal settlement as configured (accounting required).
